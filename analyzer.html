function analyze(){ let freq = {}; let dayFreq = {}; let last = null; let prev = null; let lastDay = null; let prevDay = null; // scan full record grid.forEach(row=>{ row.forEach((v,i)=>{ if(/^\d{2}$/.test(v)){ freq[v] = (freq[v] || 0) + 1; let d = headers[i]; if(!dayFreq[d]) dayFreq[d] = {}; dayFreq[d][v] = (dayFreq[d][v] || 0) + 1; prev = last; prevDay = lastDay; last = v; lastDay = d; } }); }); if(!last || !prev){ result.innerHTML = "Not enough data"; return; } // show last & prev document.getElementById("last").innerHTML = `Last: ${last} (${lastDay})
Prev: ${prev} (${prevDay})`; let famLast = family(last); let famPrev = family(prev); let score = {}; Object.keys(freq).forEach(j=>{ let s = freq[j]; let why = []; if(famLast.includes(j)){ s += 5; why.push("family of last jodi"); } if(famPrev.includes(j)){ s += 4; why.push("family of previous jodi"); } if(dayFreq[lastDay] && dayFreq[lastDay][j]){ s += dayFreq[lastDay][j] * 3; why.push("same day repeat"); } if(dayFreq[prevDay] && dayFreq[prevDay][j]){ s += dayFreq[prevDay][j] * 2; why.push("previous day effect"); } family(j).forEach(x=>{ if(famLast.includes(x) || famPrev.includes(x)){ s += 2; } }); score[j] = {val:s, why:[...new Set(why)]}; }); // 🔒 STRICT TOP 8 ONLY let top = Object.entries(score) .sort((a,b)=>b[1].val - a[1].val) .slice(0,8); let html = ""; top.forEach(([j,o],i)=>{ html += `
${i+1}. ${j}
Family: ${family(j).join(", ")}
Score: ${o.val}
🤖 ${o.why.join(" | ") || "frequency + family overlap"}
`; }); result.innerHTML = html; }
